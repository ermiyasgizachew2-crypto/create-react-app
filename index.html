Pine Script v5 LSD Code

Copy & paste into TradingView Pine Editor

////@version=5
indicator("Liquidity Shift Detection (LSD)", overlay=true)

// Inputs
leftLen  = input.int(5, "Pivot Left Length")
rightLen = input.int(5, "Pivot Right Length")

// Pivot High / Low Detection
pH = ta.pivothigh(high, leftLen, rightLen)
pL = ta.pivotlow(low, leftLen, rightLen)

var float lastPH = na
var float lastPL = na

if not na(pH)
    lastPH := pH
if not na(pL)
    lastPL := pL

// Liquidity Sweep Logic
bullSweep = not na(lastPL) and low < lastPL and close > lastPL
bearSweep = not na(lastPH) and high > lastPH and close < lastPH

// Plotting Signals on Chart
plotshape(bullSweep, title="Bullish LSD Sweep", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.small)
plotshape(bearSweep, title="Bearish LSD Sweep", style=shape.triangledown, location=location.abovebar, color=color.red, size=size.small)

alertcondition(bullSweep, title="Bullish LSD Sweep Alert", message="Bullish Liquidity Sweep detected!")
alertcondition(bearSweep, title="Bearish LSD Sweep Alert", message="Bearish Liquidity Sweep detected!")
