            // Lights
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.2);
            scene.add(ambientLight);

            const pointLight = new THREE.PointLight(0xF3BA2F, 1.5, 100);
            pointLight.position.set(10, 20, 10);
            scene.add(pointLight);

            const greenLight = new THREE.PointLight(0x00e676, 1.5, 100);
            greenLight.position.set(-10, -10, -10);
            scene.add(greenLight);

            // Structure elements (Financial grid simulation)
            const gridHelper = new THREE.GridHelper(40, 40, 0x1f2d4d, 0x0c172d);
            gridHelper.position.y = -4;
            scene.add(gridHelper);

            // Generate "Trendline Nodes" (Points)
            const geometry = new THREE.BufferGeometry();
            const count = 120;
            const positions = new Float32Array(count * 3);
            const colors = new Float32Array(count * 3);

            for (let i = 0; i < count * 3; i += 3) {
                positions[i] = (Math.random() - 0.5) * 35;
                positions[i + 1] = Math.sin(positions[i] * 0.15) * 4 + (Math.random() - 0.5) * 2;
                positions[i + 2] = (Math.random() - 0.5) * 35;

                // Color gradients (Binance Gold to Teal range)
                colors[i] = 0.95; // R
                colors[i + 1] = 0.73; // G
                colors[i + 2] = 0.18 + Math.random() * 0.2; // B
            }

            geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
            geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));

            const material = new THREE.PointsMaterial({
                size: 0.65,
                vertexColors: true,
                transparent: true,
                opacity: 0.8
            });

            const particles = new THREE.Points(geometry, material);
            scene.add(particles);

            // Generate order blocks (Glowing cubes)
            const blocksGroup = new THREE.Group();
            const boxGeo = new THREE.BoxGeometry(3, 1.5, 3);
            const boxMat1 = new THREE.MeshPhongMaterial({
                color: 0xF3BA2F,
                transparent: true,
                opacity: 0.2,
                wireframe: true
            });
            const boxMat2 = new THREE.MeshPhongMaterial({
                color: 0x00e676,
                transparent: true,
                opacity: 0.15,
                wireframe: true
            });

            const orderBlock1 = new THREE.Mesh(boxGeo, boxMat1);
            orderBlock1.position.set(-8, 2, -4);
            blocksGroup.add(orderBlock1);

            const orderBlock2 = new THREE.Mesh(boxGeo, boxMat2);
            orderBlock2.position.set(8, -1, 4);
            blocksGroup.add(orderBlock2);

            scene.add(blocksGroup);

            // Interactive mouse rotation tracking
            let isDragging = false;
            let previousMousePosition = { x: 0, y: 0 };

            container.addEventListener('mousedown', (e) => {
                isDragging = true;
            });

            container.addEventListener('mousemove', (e) => {
                const deltaMove = {
                    x: e.offsetX - previousMousePosition.x,
                    y: e.offsetY - previousMousePosition.y
                };

                if (isDragging) {
                    const deltaRotationQuaternion = new THREE.Quaternion()
                        .setFromEuler(new THREE.Euler(
                            deltaMove.y * 0.005,
                            deltaMove.x * 0.005,
                            0,
                            'XYZ'
                        ));
                    
                    particles.quaternion.multiplyQuaternions(deltaRotationQuaternion, particles.quaternion);
                    blocksGroup.quaternion.multiplyQuaternions(deltaRotationQuaternion, blocksGroup.quaternion);
                }

                previousMousePosition = {
                    x: e.offsetX,
                    y: e.offsetY
                };
            });

            window.addEventListener('mouseup', () => {
                isDragging = false;
            });

            // Handle window resizing
            window.addEventListener('resize', () => {
                const newWidth = container.clientWidth;
                const newHeight = container.clientHeight;
                camera.aspect = newWidth / newHeight;
                camera.updateProjectionMatrix();
                renderer.setSize(newWidth, newHeight);
            });

            // Animation loop
            function animate() {
                requestAnimationFrame(animate);

                // Auto rotate particles slowly
                if(!isDragging) {
                    particles.rotation.y += 0.001;
                    blocksGroup.rotation.y -= 0.001;
                }

                // Wave effect on block positions
                const time = Date.now() * 0.0015;
                orderBlock1.position.y = 2 + Math.sin(time) * 0.5;
                orderBlock2.position.y = -1 + Math.cos(time) * 0.4;

                renderer.render(scene, camera);
            }

            // Fire animation
            animate();
        };
    </script>
